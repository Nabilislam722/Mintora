#include <WiFi.h>
#include <WebServer.h>
#include <math.h>

// ====== Wi-Fi Credentials ======
const char* WIFI_SSID = "YOUR_WIFI_SSID";
const char* WIFI_PASS = "YOUR_WIFI_PASSWORD";

// ====== Hardware Pins & LEDC Config ======
const int PIN_CW = 33;  // Cool White -> 1k R3 -> Q1 BC547
const int PIN_WW = 32;  // Warm White -> 1k R3 -> Q1 BC547

uint32_t pwmFreq = 19531;               // 19.53 kHz @ 12-bit
const uint8_t PWM_RES = 12;
const uint16_t MAX_DUTY = (1 << PWM_RES) - 1; // 4095

WebServer server(80);

float currentCwPct = 0.0f;
float currentWwPct = 0.0f;
float currentGamma = 1.0f;

// Inverted 12-bit duty for Q1 (BC547) level-shifter:
//   0%   -> 4095 (GPIO HIGH -> Q1 ON  -> MOSFET OFF)
//   100% -> 0    (GPIO LOW  -> Q1 OFF -> MOSFET ON)
uint16_t pctToInvertedDuty(float pct, float gammaVal) {
  pct = constrain(pct, 0.0f, 100.0f);
  if (pct <= 0.01f) return MAX_DUTY;
  if (pct >= 99.99f) return 0;

  float norm = pct / 100.0f;
  float corrected = (gammaVal > 1.01f) ? powf(norm, gammaVal) : norm;
  uint16_t forwardDuty = (uint16_t)lroundf(corrected * MAX_DUTY);
  return MAX_DUTY - forwardDuty;
}

void sendCorsHeaders() {
  server.sendHeader("Access-Control-Allow-Origin", "*");
  server.sendHeader("Access-Control-Allow-Methods", "GET, POST, OPTIONS");
  server.sendHeader("Access-Control-Allow-Headers", "Content-Type");
}

void handleOptions() {
  sendCorsHeaders();
  server.send(204);
}

// Endpoint: /api/set?cw=50&ww=50&gamma=2.2
void handleSet() {
  sendCorsHeaders();

  if (server.hasArg("cw")) {
    currentCwPct = constrain(server.arg("cw").toFloat(), 0.0f, 100.0f);
  }
  if (server.hasArg("ww")) {
    currentWwPct = constrain(server.arg("ww").toFloat(), 0.0f, 100.0f);
  }
  if (server.hasArg("gamma")) {
    currentGamma = constrain(server.arg("gamma").toFloat(), 1.0f, 2.4f);
  }

  uint16_t dutyCw = pctToInvertedDuty(currentCwPct, currentGamma);
  uint16_t dutyWw = pctToInvertedDuty(currentWwPct, currentGamma);

  ledcWrite(PIN_CW, dutyCw);
  ledcWrite(PIN_WW, dutyWw);

  String json = "{";
  json += "\"cw\":" + String(currentCwPct, 1) + ",";
  json += "\"ww\":" + String(currentWwPct, 1) + ",";
  json += "\"gamma\":" + String(currentGamma, 2) + ",";
  json += "\"dutyCw\":" + String(dutyCw) + ",";
  json += "\"dutyWw\":" + String(dutyWw) + ",";
  json += "\"freq\":" + String(pwmFreq);
  json += "}";

  server.send(200, "application/json", json);
}

// Endpoint: /api/freq?hz=19531
void handleFreq() {
  sendCorsHeaders();

  if (server.hasArg("hz")) {
    uint32_t hz = (uint32_t)server.arg("hz").toInt();
    if (hz >= 1000 && hz <= 80000) {
      pwmFreq = hz;
      ledcChangeFrequency(PIN_CW, pwmFreq, PWM_RES);
      ledcChangeFrequency(PIN_WW, pwmFreq, PWM_RES);
      ledcWrite(PIN_CW, pctToInvertedDuty(currentCwPct, currentGamma));
      ledcWrite(PIN_WW, pctToInvertedDuty(currentWwPct, currentGamma));
    }
  }

  server.send(200, "application/json", "{\"freq\":" + String(pwmFreq) + "}");
}

void setup() {
  Serial.begin(115200);

  ledcAttach(PIN_CW, pwmFreq, PWM_RES);
  ledcAttach(PIN_WW, pwmFreq, PWM_RES);

  // Start safely OFF (duty = 4095 -> GPIO HIGH)
  ledcWrite(PIN_CW, MAX_DUTY);
  ledcWrite(PIN_WW, MAX_DUTY);

  WiFi.mode(WIFI_STA);
  WiFi.begin(WIFI_SSID, WIFI_PASS);
  Serial.print("Connecting to Wi-Fi");
  while (WiFi.status() != WL_CONNECTED) {
    delay(400);
    Serial.print(".");
  }
  Serial.println();
  Serial.print("ESP32 IP Address: http://");
  Serial.println(WiFi.localIP());

  server.on("/api/set", HTTP_OPTIONS, handleOptions);
  server.on("/api/set", HTTP_GET, handleSet);
  server.on("/api/set", HTTP_POST, handleSet);

  server.on("/api/freq", HTTP_OPTIONS, handleOptions);
  server.on("/api/freq", HTTP_GET, handleFreq);
  server.on("/api/freq", HTTP_POST, handleFreq);

  server.begin();
}

void loop() {
  server.handleClient();
}
